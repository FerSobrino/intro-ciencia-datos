% PORTADA (fondo negro, título en bugambilia — sin pentagramas)
% ============================================================
\makeatletter
\setbeamertemplate{title page}{%
  \begin{tikzpicture}[remember picture, overlay]
    \fill[BgBlack] (current page.south west) rectangle (current page.north east);
    \draw[BgPink!70, line width=0.6pt]
      ([xshift=0.9cm, yshift=-0.9cm]current page.north west) --
      ([xshift=-0.9cm, yshift=-0.9cm]current page.north east);
    \draw[BgDimPink, line width=0.6pt]
      ([xshift=0.9cm, yshift=0.9cm]current page.south west) --
      ([xshift=-0.9cm, yshift=0.9cm]current page.south east);
  \end{tikzpicture}%
  \vspace{1.5cm}
  \begin{center}
    {\fontsize{26}{36}\selectfont\titlefont\color{BgPink}\MakeUppercase{\inserttitle}}\\[8pt]
    \ifx\insertsubtitle\@empty\else
      {\fontsize{11}{14}\selectfont\color{BgWhite}\insertsubtitle}\\[12pt]
    \fi
    {\color{BgDimPink}\rule{0.52\textwidth}{0.45pt}}\\[10pt]
    {\small\color{BgGray}\insertauthor}\\[2pt]
    {\small\color{BgGray}\insertdate}
  \end{center}
}
\makeatother
